function progressStats(){const s=store(),q=qstore();let tot=0,done=0,topics=0,topicsDone=0,qTaken=0,qRight=0,qTot=0;
  C.forEach(c=>{const m=microTopics(c);if(!m.length)return;topics++;let d=0;m.forEach((_,i)=>{tot++;if(s['mt:'+c.id+':'+i]){done++;d++}});if(d===m.length)topicsDone++;
    if(QUIZ[c.id]){if(q[c.id]!=null){qTaken++;qRight+=q[c.id];qTot+=QUIZ[c.id].length}}});
  return {tot,done,topics,topicsDone,qTaken,qRight,qTot,quizTopics:Object.keys(QUIZ).length};}
function renderProgress(){document.querySelectorAll('.prog').forEach(box=>{const st=progressStats();const pct=st.tot?Math.round(100*st.done/st.tot):0;box.innerHTML='';
  const h=document.createElement('h4');h.textContent='My progress (only you)';const p=document.createElement('div');p.className='pct';p.textContent=pct+'%';
  const bar=document.createElement('div');bar.className='pbar';const fill=document.createElement('span');fill.style.width=pct+'%';bar.appendChild(fill);
  const s=document.createElement('div');s.className='pstats';

  const row=document.createElement('div');row.className='prow';
  const rst=document.createElement('button');rst.type='button';rst.textContent='Reset';rst.onclick=()=>{if(rst.dataset.arm){saveStore({});qsave({});rst.dataset.arm='';renderProgress();if(!document.getElementById('syllabusView').hidden)renderSyllabus();else if(cur)load(cur.id,false);progMsg('Your progress was cleared.');}else{rst.dataset.arm='1';rst.textContent='Click again to clear';setTimeout(()=>{rst.dataset.arm='';rst.textContent='Reset'},4000);}};

function exportProgress(){const data={profile:curProfile(),exported:new Date().toISOString(),done:store(),quiz:qstore()};
  const a=document.createElement('a');a.href=URL.createObjectURL(new Blob([JSON.stringify(data,null,1)],{type:'application/json'}));a.download='fde-atlas-progress.json';document.body.appendChild(a);a.click();setTimeout(()=>{URL.revokeObjectURL(a.href);a.remove()},500);}
function importProgress(){const inp=document.createElement('input');inp.type='file';inp.accept='.json,application/json';inp.onchange=()=>{const f=inp.files[0];if(!f)return;const r=new FileReader();
  r.onload=()=>{try{const d=JSON.parse(r.result);if(typeof d.done!=='object'||typeof d.quiz!=='object')throw 0;saveStore(Object.assign(store(),d.done));qsave(Object.assign(qstore(),d.quiz));renderProgress();if(cur)load(cur.id,false);progMsg('Progress loaded from file.');}catch(e){progMsg('That file is not an FDE Atlas progress backup.')}};r.readAsText(f);};inp.click();}

function lblPos(a,b,mid,label,off){const dx=Math.abs(b[0]-a[0]),dy=Math.abs(b[1]-a[1]),lw=label.length*6.4;
  if(dy<12&&lw+10>dx-W)return mid[1]>a[1]+1?[mid[0],Math.max(a[1],b[1])+H/2+15,'middle']:[mid[0],Math.min(a[1],b[1])-H/2-7,'middle'];
  if(dx<12&&dy-H<24)return [mid[0]+8,mid[1]+4,'start'];
  return [mid[0],mid[1]+off,'middle'];}
