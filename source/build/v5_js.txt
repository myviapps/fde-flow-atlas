function mdInline(s){return s.replace(/`([^`]+)`/g,'<code>$1</code>').replace(/\*\*([^*]+)\*\*/g,'<strong>$1</strong>').replace(/(^|[^*\w])\*([^*\s][^*]*)\*/g,'$1<em>$2</em>').replace(/(^|\W)_([^_\s][^_]*)_(?=\W|$)/g,'$1<em>$2</em>').replace(/\[([^\]]+)\]\([^)]*\)/g,'<u>$1</u>');}
function mdToHtml(md){const esc=t=>t.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');const L=md.replace(/\r/g,'').split('\n');let out='',i=0;
  while(i<L.length){let l=L[i];
    if(/^```/.test(l)){let code=[];i++;while(i<L.length&&!/^```/.test(L[i]))code.push(L[i++]);i++;out+='<pre><code>'+esc(code.join('\n'))+'</code></pre>';continue;}
    let m=l.match(/^(#{1,4})\s+(.*)/);if(m){const n=Math.min(m[1].length,3);out+='<h'+n+'>'+mdInline(esc(m[2]))+'</h'+n+'>';i++;continue;}
    if(/^\s*(---|\*\*\*)\s*$/.test(l)){out+='<hr>';i++;continue;}
    if(/^\s*\|/.test(l)){let rows=[];while(i<L.length&&/^\s*\|/.test(L[i]))rows.push(L[i++]);
      const cells=r=>r.trim().replace(/^\||\|$/g,'').split('|').map(x=>mdInline(esc(x.trim())));let h='<table>';
      rows.forEach((r,ri)=>{if(/^\s*\|?\s*:?-{2,}/.test(r))return;const t=ri===0?'th':'td';h+='<tr>'+cells(r).map(x=>'<'+t+'>'+x+'</'+t+'>').join('')+'</tr>';});out+=h+'</table>';continue;}
    if(/^\s*>/.test(l)){let q=[];while(i<L.length&&/^\s*>/.test(L[i]))q.push(L[i++].replace(/^\s*>\s?/,''));out+='<blockquote>'+mdInline(esc(q.join(' ')))+'</blockquote>';continue;}
    if(/^\s*([-*+]|\d+[.)])\s+/.test(l)){const ord=/^\s*\d/.test(l);let items='';
      while(i<L.length&&/^\s*([-*+]|\d+[.)])\s+/.test(L[i])){const ind=L[i].match(/^\s*/)[0].length;let t=L[i].replace(/^\s*([-*+]|\d+[.)])\s+/,'');
        t=esc(t).replace(/^\[ \]\s*/,'&#9744; ').replace(/^\[x\]\s*/i,'&#9745; ');i++;
        while(i<L.length&&/^\s{2,}\S/.test(L[i])&&!/^\s*([-*+]|\d+[.)])\s+/.test(L[i]))t+=' '+esc(L[i++].trim());
        items+='<li'+(ind>=2?' style="margin-left:'+Math.min(ind,8)*10+'px"':'')+'>'+mdInline(t)+'</li>';}
      out+=ord?'<ol>'+items+'</ol>':'<ul>'+items+'</ul>';continue;}
    if(!l.trim()){i++;continue;}
    let p=[];while(i<L.length&&L[i].trim()&&!/^(#{1,4}\s|```|\s*\||\s*>|\s*([-*+]|\d+[.)])\s+|\s*---\s*$)/.test(L[i]))p.push(L[i++]);out+='<p>'+mdInline(esc(p.join(' ')))+'</p>';}
  return out;}

function tplCopy(){if(!curTpl)return;const m=document.getElementById('tplmsg');const ok=()=>m.textContent='Copied.';
  const fb=()=>{const t=document.createElement('textarea');t.value=curTpl.md;document.body.appendChild(t);t.select();try{document.execCommand('copy');ok()}catch(e){m.textContent='Copy failed, select the text instead.'}t.remove();};
  try{navigator.clipboard.writeText(curTpl.md).then(ok,fb)}catch(e){fb()}}
